    observation(eyes_closed),

    observation(hand_near_face),
    observation(looking_down).

attentive_student :-
    observation(eye_contact),
    observation(upright_posture),
    \+ observation(eyes_closed),
    \+ observation(phone_suspected).

attention_level(high) :- attentive_student.
